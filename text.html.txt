    <script>
        // State variables
        let currentMood = '';
        let ratingsState = {1:0, 2:0, 3:0, 4:0};

        // Initialize draft on load
        window.onload = () => {
            loadDraft();
            // Set today's date if empty
            if(!document.getElementById('journalDate').value) {
                const today = new Date().toISOString().split('T')[0];
                document.getElementById('journalDate').value = today;
                updateDayDisplay();
            }
        };

        function updateDayDisplay() {
            const dateVal = document.getElementById('journalDate').value;
            if(!dateVal) return;
            const days = ['Ahad', 'Isnin', 'Selasa', 'Rabu', 'Khamis', 'Jumaat', 'Sabtu'];
            const dayName = days[new Date(dateVal).getDay()];
            document.getElementById('currentDayDisplay').innerHTML = `<i class="fa-regular fa-clock mr-1 text-pastel-purple"></i> Hari: ${dayName}`;
            return dayName;
        }

        function selectMood(mood) {
            currentMood = mood;
            document.querySelectorAll('.cloud-box').forEach(el => {
                el.classList.remove('active', 'border-pastel-purple', 'bg-[#FFF5F7]');
            });
            document.getElementById('mood-'+mood).classList.add('active', 'border-pastel-purple', 'bg-[#FFF5F7]');
        }

        function rateTask(taskId, stars) {
            ratingsState[taskId] = stars;
            const container = document.querySelector(`.star-rating[data-task="${taskId}"]`);
            const icons = container.querySelectorAll('.star-icon');
            icons.forEach((icon, index) => {
                if(index < stars) {
                    icon.classList.remove('fa-regular');
                    icon.classList.add('fa-solid');
                } else {
                    icon.classList.remove('fa-solid');
                    icon.classList.add('fa-regular');
                }
            });
        }

        function getFormData() {
            return {
                name: document.getElementById('childName').value,
                age: document.getElementById('childAge').value,
                date: document.getElementById('journalDate').value,
                day: updateDayDisplay() || '-',
                mood: currentMood,
                mathJenis: document.getElementById('mathJenis').value,
                mathJumlah: document.getElementById('mathJumlah').value,
                mathMarkah: document.getElementById('mathMarkah').value,
                wsJenis: document.getElementById('wsJenis').value,
                wsJumlah: document.getElementById('wsJumlah').value,
                stJenis: document.getElementById('stJenis').value,
                stJumlah: document.getElementById('stJumlah').value,
                solat: {
                    subuh: document.getElementById('solatSubuh').checked,
                    zohor: document.getElementById('solatZohor').checked,
                    asar: document.getElementById('solatAsar').checked,
                    maghrib: document.getElementById('solatMaghrib').checked,
                    isyak: document.getElementById('solatIsyak').checked
                },
                quranSurah: document.getElementById('quranSurah').value,
                quranAyat: document.getElementById('quranAyat').value,
                subMatematik: document.getElementById('subMatematik').value,
                subBahasa: document.getElementById('subBahasa').value,
                subSains: document.getElementById('subSains').value,
                subKreatif: document.getElementById('subKreatif').value,
                checkboxes: {
                    1: document.getElementById('check-1').checked,
                    2: document.getElementById('check-2').checked,
                    3: document.getElementById('check-3').checked,
                    4: document.getElementById('check-4').checked
                },
                ratings: ratingsState,
                parentNotes: document.getElementById('parentNotes').value
            };
        }

        function saveJournal() {
            const data = getFormData();
            localStorage.setItem('nprecious_journal', JSON.stringify(data));
            
            renderPdfPreviewTemplate(data);
            
            // Celebration effects
            confetti({ particleCount: 100, spread: 70, origin: { y: 0.6 }, colors: ['#FFC8DD', '#B5E2FA', '#FFF3B0'] });
            
            // Swap visibility
            document.getElementById('inputFormContainer').classList.add('hidden');
            document.getElementById('previewBox').classList.remove('hidden');
            
            window.scrollTo({ top: 0, behavior: 'smooth' });
            showToast('Jurnal disimpan & sedia untuk dicetak!');
        }

        function editJournal() {
            // Swap visibility back
            document.getElementById('previewBox').classList.add('hidden');

            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        function renderPdfPreviewTemplate(data) {
            // Update Header Info
            document.getElementById('pdfName').innerText = data.name || '-';
            document.getElementById('pdfAge').innerText = data.age || '-';
            
            // Format Tarikh kepada dd/mm/yyyy (hb:mth:year)
            if (data.date) {
                const parts = data.date.split('-');
                if(parts.length === 3) {
                    document.getElementById('pdfDate').innerText = `${parts[2]}/${parts[1]}/${parts[0]}`;
                } else {
                    document.getElementById('pdfDate').innerText = data.date;
                }
            } else {
                document.getElementById('pdfDate').innerText = '-';
            }

            document.getElementById('pdfDay').innerText = data.day || '-';
            document.getElementById('pdfMood').innerText = data.mood || '-';

            // Update AI & Kraf
            document.getElementById('pdfMathJenis').innerText = data.mathJenis || '-';
            document.getElementById('pdfMathJumlah').innerText = data.mathJumlah || '-';
            document.getElementById('pdfMathMarkah').innerText = data.mathMarkah || '-';
            
            document.getElementById('pdfWsJenis').innerText = data.wsJenis || '-';
            document.getElementById('pdfWsJumlah').innerText = data.wsJumlah || '-';
            
            document.getElementById('pdfStJenis').innerText = data.stJenis || '-';
            document.getElementById('pdfStJumlah').innerText = data.stJumlah || '-';

            // Update Solat & Al-Quran
            const solatKeys = ['subuh', 'zohor', 'asar', 'maghrib', 'isyak'];
            solatKeys.forEach(key => {
                const el = document.getElementById(`pdfSolat${key.charAt(0).toUpperCase() + key.slice(1)}`);
                if(data.solat && data.solat[key]) {
                    el.className = "fa-solid fa-square-check text-emerald-600 text-sm";
                } else {
                    el.className = "fa-regular fa-square text-gray-300 text-sm";
                }
            });

            document.getElementById('pdfQuranSurah').innerText = data.quranSurah || '-';
            document.getElementById('pdfQuranAyat').innerText = data.quranAyat || '-';

            // Update Subjek Pembelajaran
            document.getElementById('pdfSubMatematik').innerText = data.subMatematik || '-';
            document.getElementById('pdfSubBahasa').innerText = data.subBahasa || '-';
            document.getElementById('pdfSubSains').innerText = data.subSains || '-';
            document.getElementById('pdfSubKreatif').innerText = data.subKreatif || '-';

            // Update Checkboxes & Stars
            for(let i=1; i<=4; i++) {
                // Checkbox Icons
                const checkIcon = document.getElementById(`pdfCheck${i}`);
                if(data.checkboxes && data.checkboxes[i]) {
                    checkIcon.className = "fa-solid fa-square-check text-green-500";
                } else {
                    checkIcon.className = "fa-regular fa-square text-gray-300";
                }

                // Star Rating Generation
                const starCount = (data.ratings && data.ratings[i]) ? data.ratings[i] : 0;
                let starString = '';
                for(let s=0; s<5; s++) {
                    if(s < starCount) starString += '★';
                    else starString += '☆';
                }
                document.getElementById(`pdfStar${i}`).innerText = starString;
            }

            // Update Nota Ibu Bapa
            document.getElementById('pdfParentNotes').innerText = data.parentNotes || '-';
        }

        function loadDraft() {
            try {
                const saved = localStorage.getItem('nprecious_journal');
                if(!saved) return;
                
                const data = JSON.parse(saved);
                
                if(data.name) document.getElementById('childName').value = data.name;
                if(data.age) document.getElementById('childAge').value = data.age;
                if(data.date) document.getElementById('journalDate').value = data.date;
                
                if(data.mood) selectMood(data.mood);
                
                if(data.mathJenis) document.getElementById('mathJenis').value = data.mathJenis;
                if(data.mathJumlah) document.getElementById('mathJumlah').value = data.mathJumlah;
                if(data.mathMarkah) document.getElementById('mathMarkah').value = data.mathMarkah;

                if(data.wsJenis) document.getElementById('wsJenis').value = data.wsJenis;
                if(data.wsJumlah) document.getElementById('wsJumlah').value = data.wsJumlah;
                
                if(data.stJenis) document.getElementById('stJenis').value = data.stJenis;
                if(data.stJumlah) document.getElementById('stJumlah').value = data.stJumlah;

                if(data.solat) {
                    if(data.solat.subuh) document.getElementById('solatSubuh').checked = true;
                    if(data.solat.zohor) document.getElementById('solatZohor').checked = true;
                    if(data.solat.asar) document.getElementById('solatAsar').checked = true;
                    if(data.solat.maghrib) document.getElementById('solatMaghrib').checked = true;
                    if(data.solat.isyak) document.getElementById('solatIsyak').checked = true;
                }

                if(data.quranSurah) document.getElementById('quranSurah').value = data.quranSurah;
                if(data.quranAyat) document.getElementById('quranAyat').value = data.quranAyat;

                if(data.subMatematik) document.getElementById('subMatematik').value = data.subMatematik;
                if(data.subBahasa) document.getElementById('subBahasa').value = data.subBahasa;
                if(data.subSains) document.getElementById('subSains').value = data.subSains;
                if(data.subKreatif) document.getElementById('subKreatif').value = data.subKreatif;

                if(data.parentNotes) document.getElementById('parentNotes').value = data.parentNotes;

                if(data.checkboxes) {
                    for(let i=1; i<=4; i++) {
                        if(data.checkboxes[i]) document.getElementById(`check-${i}`).checked = true;
                    }
                }
                
                if(data.ratings) {
                    for(let i=1; i<=4; i++) {
                        rateTask(i, data.ratings[i]);
                    }
                }

                if(data.date) updateDayDisplay();

            } catch(e) {
                console.error("Gagal memuatkan draf: ", e);
            }
        }

        function showToast(message) {
            const toast = document.createElement('div');
            toast.className = 'fixed bottom-5 right-5 bg-pastel-purple text-white px-5 py-3 rounded-2xl shadow-lg font-bold text-sm transition-all duration-300 z-50 flex items-center gap-2';
            toast.innerHTML = `<i class="fa-solid fa-circle-check"></i> ${message}`;
            document.body.appendChild(toast);
            setTimeout(() => {
                toast.style.opacity = '0';
                setTimeout(() => toast.remove(), 300);
            }, 2500);
        }
